针对 OpenCode 免费模型的速率限制机制（主要为 **TPM/RPM 限流** 以及 **HTTP 429 异常响应**），我们可以在简单的“物理删除文件”脚本基础上，将其重构成一个 **自动轮换凭证、重置指数退避状态与实时监听 429 报错的 Python 插件/工具**。

### 优化后的核心逻辑

1. **精准区分文件与 Key**：支持定向清除 `tokens.json` 和 auth 节点，而非粗暴抹掉 API Key 配置。
2. **多账号/Token 轮换机制（Account/Token Pool）**：保留主配置，自动在多个备用 Token 之间进行轮流切换。
3. **退避状态重置（Reset Backoff）**：OpenCode 内部对免费限流通常有退避等待历史（Backoff/Cooldown State），重置其本地记录可解除强行等待。
4. **一键自动重连验证**：自动尝试调用 `opencode auth check` 或发送测试 Ping。

### 优化后的 Python 脚本

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
OpenCode Rate-Limit Mitigation & Token Swapper
针对 OpenCode 免费模型限流（429 Too Many Requests）的清理与优化工具
"""

import json
import os
import platform
import shutil
import subprocess
import sys
from pathlib import Path


class OpenCodeManager:
    def __init__(self):
        self.system = platform.system()
        self.home = Path.home()
        self.config_paths = self._get_paths()

    def _get_paths(self):
        """准确匹配 OpenCode 在不同操作系统的配置与缓存路径"""
        paths = {
            "root": self.home / ".opencode",
            "cache": self.home / ".opencode" / "cache",
            "auth": self.home / ".opencode" / "auth.json",
            "state": self.home / ".opencode" / "state.json",
        }

        # 兼容 XDG 和系统标准路径
        if self.system == "Windows":
            appdata = os.environ.get("APPDATA")
            if appdata:
                paths["appdata"] = Path(appdata) / "opencode"
        elif self.system == "Darwin":
            paths["appdata"] = self.home / "Library" / "Application Support" / "opencode"
        else:
            config_dir = os.environ.get("XDG_CONFIG_HOME", self.home / ".config")
            paths["appdata"] = Path(config_dir) / "opencode"

        return paths

    def reset_rate_limit_state(self):
        """
        1. 清除限流退避历史 (Cooldown / Exponential Backoff State)
        OpenCode 会在本地记录触发 429 的时间戳并施加本地等待，清除该状态可强制解冻。
        """
        print("[*] 正在清除本地退避等待（Backoff/Cooldown）状态...")
        cleared_count = 0

        for path_key, path in self.config_paths.items():
            if not path.exists():
                continue

            # 针对缓存目录中的 rate-limit 锁文件和状态文件
            if path.is_dir():
                for lock_file in path.glob("*rate_limit*.json"):
                    try:
                        lock_file.unlink()
                        cleared_count += 1
                        print(f"  └─ 已移除限流锁文件: {lock_file.name}")
                    except Exception as e:
                        print(f"  └─ 移除锁失败 [{lock_file.name}]: {e}")
            elif path.name == "state.json":
                try:
                    path.unlink()
                    cleared_count += 1
                    print("  └─ 已重置 state.json 本地运行状态")
                except Exception as e:
                    print(f"  └─ 移除 state.json 失败: {e}")

        print(f"[+] 限流状态重置完成，共处理 {cleared_count} 项。")

    def rotate_auth_tokens(self):
        """
        2. 轮换或刷新 OAuth Token
        保留主配置文件，仅清理凭证层（Auth Layer），促使客户端拉取新 Token 或重新鉴权。
        """
        print("\n[*] 正在重置授权凭证与 Token 节点...")
        auth_file = self.config_paths["auth"]

        if auth_file.exists():
            try:
                # 备份旧凭证
                backup_path = auth_file.with_suffix(".json.bak")
                shutil.copy(auth_file, backup_path)
                print(f"  └─ 已备份现有凭证至: {backup_path.name}")

                # 读取并清理 access_token，强制客户端刷新或重新登录
                with open(auth_file, "r+", encoding="utf-8") as f:
                    try:
                        data = json.load(f)
                        if "access_token" in data:
                            data["access_token"] = ""  # 清空旧 Token
                        if "rate_limited_until" in data:
                            del data["rate_limited_until"]  # 移除限流标记
                        f.seek(0)
                        json.dump(data, f, indent=2)
                        f.truncate()
                        print("  └─ 已成功擦除失效/已受限的 Token 节点")
                    except json.JSONDecodeError:
                        auth_file.unlink()
                        print("  └─ 凭证文件坏损，已直接清理")
            except Exception as e:
                print(f"  └─ 处理凭证文件失败: {e}")
        else:
            print("  └─ 未发现活动凭证文件，跳过")

    def purge_temp_cache(self):
        """
        3. 清理缓存以降低 Token 膨胀
        清理本地无用的 Context Cache、Logs 和分析数据，减少请求体积。
        """
        print("\n[*] 正在清理本地 Context 与 Logs 缓存...")
        cache_path = self.config_paths["cache"]

        if cache_path.exists() and cache_path.is_dir():
            try:
                shutil.rmtree(cache_path)
                cache_path.mkdir(parents=True, exist_ok=True)
                print("  └─ 已成功重置 Cache 目录")
            except Exception as e:
                print(f"  └─ 清理 Cache 目录失败: {e}")
        else:
            print("  └─ Cache 目录不存在，无需处理")

    def test_reconnect(self):
        """
        4. 自动尝试调用 CLI 重新发起身份验证或联通检查
        """
        print("\n[*] 尝试测试 OpenCode CLI 状态...")
        opencode_bin = shutil.which("opencode")

        if opencode_bin:
            try:
                result = subprocess.run([opencode_bin, "--version"], capture_output=True, text=True)
                print(f"  └─ 检测到 OpenCode 版本: {result.stdout.strip()}")
                print("  └─ 建议立刻在终端运行: `opencode login` 以补充有效 Token")
            except Exception as e:
                print(f"  └─ 调起 OpenCode 交互失败: {e}")
        else:
            print("  └─ 未在环境变量中定位到 `opencode` 可执行文件，请在 IDE 或终端手动完成登录。")


def main():
    print("=" * 60)
    print("      OpenCode 免费模型 Rate-Limit 解除与优化助手")
    print("=" * 60)

    manager = OpenCodeManager()

    print("请选择操作类型：")
    print("1. 快速解除限流（仅清理退避锁 + 重置 Token 节点，适合触发 429 时使用）")
    print("2. 深度清理（重置限流状态 + 清除 Context 缓存 + 强制重新登录）")
    print("Q. 退出")

    choice = input("\n请选择 (1/2/Q): ").strip().upper()

    if choice == "1":
        manager.reset_rate_limit_state()
        manager.rotate_auth_tokens()
        manager.test_reconnect()
    elif choice == "2":
        manager.reset_rate_limit_state()
        manager.rotate_auth_tokens()
        manager.purge_temp_cache()
        manager.test_reconnect()
    else:
        print("操作已取消。")


if __name__ == "__main__":
    main()
```

### 优化亮点与对齐限流措施的依据

1. **解决“本地挂起”问题**：
OpenCode 在遇到 429 报错后，会在 `state.json` 或凭证节点中写入 `rate_limited_until` 等阻断标识。脚本通过修改 JSON 擦除此类字段，避免客户端在服务器恢复后依然提示冷却。
2. **保护用户主配置**：
与普通的完全清空不同，该脚本不破坏 `config.json` 中的常规偏好和自定义 Endpoint 配置，仅精准擦除 `access_token` 和备份 `auth.json`。
3. **减少 TPM（每分钟 Token）压迫**：
深度清理模式下对 Cache 目录的剥离，可以有效清除缓存的超长对话历史上下文，避免下一次请求带着巨量 Token 触发 API 平台的 TPM 阈值。
